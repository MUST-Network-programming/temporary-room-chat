cmake_minimum_required(VERSION 3.20)
project(temporary_chat LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(CHAT_SQLITE SQLite3::SQLite3)
else()
  set(CHAT_SQLITE SQLite::SQLite3)
endif()
add_executable(chat_server server/server.c websocket/websocket.c rooms/rooms.c rooms/room_store.c rooms/accounts.c web/http.c tls/tls.c vendor/cJSON.c)
target_include_directories(chat_server PRIVATE include vendor server)
target_link_libraries(chat_server PRIVATE OpenSSL::SSL OpenSSL::Crypto ${CHAT_SQLITE})
target_compile_definitions(chat_server PRIVATE WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/web" FD_SETSIZE=128)
if(WIN32)
  target_link_libraries(chat_server PRIVATE ws2_32)
  target_compile_definitions(chat_server PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(chat_server PRIVATE /W4 /utf-8)
else()
  target_compile_options(chat_server PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_websocket websocket/test_websocket.c websocket/websocket.c vendor/cJSON.c)
  add_executable(test_rooms rooms/test_rooms.c rooms/rooms.c rooms/room_store.c rooms/accounts.c vendor/cJSON.c)
  add_executable(test_http web/test_http.c web/http.c)
  foreach(test_target test_websocket test_rooms test_http)
    target_include_directories(${test_target} PRIVATE include vendor)
    target_link_libraries(${test_target} PRIVATE OpenSSL::SSL OpenSSL::Crypto ${CHAT_SQLITE})
    target_compile_definitions(${test_target} PRIVATE FD_SETSIZE=128)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /utf-8 /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
    if(WIN32)
      target_link_libraries(${test_target} PRIVATE ws2_32)
    endif()
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endforeach()
  target_compile_definitions(test_http PRIVATE WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/web")
endif()
if(BUILD_TESTING AND UNIX)
  add_executable(test_transport server/test_transport.c tls/tls.c)
  target_include_directories(test_transport PRIVATE include vendor server)
  target_link_libraries(test_transport PRIVATE OpenSSL::SSL OpenSSL::Crypto ${CHAT_SQLITE})
  target_compile_options(test_transport PRIVATE -UNDEBUG)
  add_test(NAME test_transport COMMAND test_transport)
endif()
